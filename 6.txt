import numpy as np

# A. Setting up the environment

class TicTacToeEnvironment:

    def __init__(self):
        self.state = [0] * 9
        self.is_terminal = False

    def reset(self):
        self.state = [0] * 9
        self.is_terminal = False

    def get_available_moves(self):
        return [i for i, mark in enumerate(self.state) if mark == 0]

    def make_move(self, move, player_mark):
        self.state[move] = player_mark

    def check_win(self, player_mark):
        winning_states = [
            [0, 1, 2], [3, 4, 5], [6, 7, 8],
            [0, 3, 6], [1, 4, 7], [2, 5, 8],
            [0, 4, 8], [2, 4, 6]
        ]

        for state_indices in winning_states:
            if all(self.state[i] == player_mark for i in state_indices):
                self.is_terminal = True
                return True

        return False

    def is_draw(self):
        return 0 not in self.state


# B. Building the Q-Learning Agent

class QLearningAgent:

    def __init__(self, learning_rate=0.9, discount_factor=0.9,
                 exploration_rate=0.3):

        self.learning_rate = learning_rate
        self.discount_factor = discount_factor
        self.exploration_rate = exploration_rate
        self.q_table = np.zeros((3**9, 9))

    def get_state_index(self, state):
        state_index = 0

        for i, mark in enumerate(state):
            state_index += (3 ** i) * (mark + 1)

        return state_index

    def choose_action(self, state, available_moves):
        state_index = self.get_state_index(state)

        if np.random.random() < self.exploration_rate:
            return np.random.choice(available_moves)

        best_move_index = np.argmax(
            self.q_table[state_index, available_moves]
        )

        return available_moves[best_move_index]

    def update_q_table(self, state, action, next_state, reward,
                       next_available_moves=None):

        state_index = self.get_state_index(state)

        if next_state is not None and len(next_available_moves) > 0:
            next_state_index = self.get_state_index(next_state)
            max_q_value = np.max(
                self.q_table[next_state_index, next_available_moves]
            )
        else:
            max_q_value = 0

        self.q_table[state_index, action] = (
            (1 - self.learning_rate) * self.q_table[state_index, action]
            + self.learning_rate *
            (reward + self.discount_factor * max_q_value)
        )


# C. Training the agents

def evaluate_agents(agent1, agent2, num_episodes):

    environment = TicTacToeEnvironment()

    agent1_wins = 0
    agent2_wins = 0
    draws = 0

    for _ in range(num_episodes):

        environment.reset()
        current_agent = agent1

        previous_moves = {
            agent1: None,
            agent2: None
        }

        while not environment.is_terminal:

            available_moves = environment.get_available_moves()
            current_state = environment.state.copy()

            if previous_moves[current_agent] is not None:
                prev_state, prev_action = previous_moves[current_agent]

                current_agent.update_q_table(
                    prev_state, prev_action, current_state,
                    0, available_moves
                )

                previous_moves[current_agent] = None

            action = current_agent.choose_action(
                current_state, available_moves
            )

            if current_agent == agent1:
                player_mark = 1
                opponent = agent2
            else:
                player_mark = -1
                opponent = agent1

            environment.make_move(action, player_mark)

            if environment.check_win(player_mark):

                current_agent.update_q_table(
                    current_state, action, None, 10
                )

                if previous_moves[opponent] is not None:
                    prev_state, prev_action = previous_moves[opponent]

                    opponent.update_q_table(
                        prev_state, prev_action, None, -10
                    )

                if current_agent == agent1:
                    agent1_wins += 1
                else:
                    agent2_wins += 1

                break

            elif environment.is_draw():

                current_agent.update_q_table(
                    current_state, action, None, 0
                )

                if previous_moves[opponent] is not None:
                    prev_state, prev_action = previous_moves[opponent]

                    opponent.update_q_table(
                        prev_state, prev_action, None, 0
                    )

                draws += 1
                break

            previous_moves[current_agent] = (current_state, action)
            current_agent = opponent

    return agent1_wins, agent2_wins, draws


# D. Training the model

agent1 = QLearningAgent()
agent2 = QLearningAgent()

num_episodes = int(input("Enter number of training games: "))

agent1_wins, agent2_wins, draws = evaluate_agents(
    agent1, agent2, num_episodes
)

print("\nTraining Results")
print("Agent 1 wins:", agent1_wins)
print("Agent 2 wins:", agent2_wins)
print("Draws:", draws)


# E. Testing the trained model against a human

def test_agent(agent):

    environment = TicTacToeEnvironment()
    environment.reset()

    agent.exploration_rate = 0

    print("\nYou are X (1), Agent is O (-1)")
    print("Enter positions from 1 to 9")

    while not environment.is_terminal:

        print("\nCurrent Board:")
        for i in range(0, 9, 3):
            print(environment.state[i:i+3])

        available_moves = environment.get_available_moves()

        # Human move
        move = int(input("Enter your move (1-9): ")) - 1

        if move not in available_moves:
            print("Invalid move! Try again.")
            continue

        environment.make_move(move, 1)

        if environment.check_win(1):
            print("You win!")
            break

        if environment.is_draw():
            print("Match Draw!")
            break

        # Agent move
        available_moves = environment.get_available_moves()
        state = environment.state.copy()

        action = agent.choose_action(state, available_moves)
        environment.make_move(action, -1)

        print("Agent selected position:", action + 1)

        if environment.check_win(-1):
            print("Agent wins!")
            break

        if environment.is_draw():
            print("Match Draw!")
            break

    print("\nFinal Board:")
    for i in range(0, 9, 3):
        print(environment.state[i:i+3])
